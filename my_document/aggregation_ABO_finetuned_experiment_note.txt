conda activate /speed-scratch/qiaoyu/env/trellis
# =========================================================
# Goal:
#   使用 ABO 上 finetune 的 encoder+decoder（170000 steps）
#   在 Toys4k_small 测试集上做编码、解码和评估。
# =========================================================

# ---------------------------------------------------------
# A. 路径约定（全部使用绝对路径）
# ---------------------------------------------------------

# 检查 checkpoint 和测试特征目录
ls -lh "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/outputs/finetune_vis_abo/ckpts/encoder_ema0.9999_step0170000.pt"
ls -lh "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/outputs/finetune_vis_abo/ckpts/decoder_ema0.9999_step0170000.pt"
ls -lh "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/datasets/Toys4k_small/features"


# ---------------------------------------------------------
# B. 用 ABO finetuned encoder 编码 Toys4k_small
# ---------------------------------------------------------
cd /nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS

python dataset_toolkits/encode_latent.py \
  --output_dir "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/datasets/Toys4k_small" \
  --feat_model dinov2_vitl14_reg \
  --feature_subdir dinov2_vitl14_reg_vis_weighted \
  --enc_ckpt_path "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/outputs/finetune_vis_abo/ckpts/encoder_ema0.9999_step0170000.pt" \
  --enc_model finetune_vis_abo \
  --model_root "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/outputs" \
  --latent_subdir dinov2_vitl14_reg_vis_weighted_abo_ft_step170k

# 写回 metadata（如果你 build_metadata 的数据集 key 不是 Toys4k，请替换）
python dataset_toolkits/build_metadata.py Toys4k --output_dir "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/datasets/Toys4k_small"


# ---------------------------------------------------------
# C. 用 ABO finetuned decoder 解码
# ---------------------------------------------------------
python my_scripts/decode_latents_to_gaussian.py \
  --output_dir "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/datasets/Toys4k_small" \
  --latent_model dinov2_vitl14_reg_vis_weighted_abo_ft_step170k \
  --dec_ckpt_path "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/outputs/finetune_vis_abo/ckpts/decoder_ema0.9999_step0170000.pt" \
  --dec_model_dir "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/outputs/finetune_vis_abo" \
  --save_dir gaussians_decoded_vis_weighted_abo_ft_step170k


# ---------------------------------------------------------
# D. 评估
# ---------------------------------------------------------
python my_scripts/eval_render_metrics.py \
  --output_dir "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/datasets/Toys4k_small" \
  --gauss_dir "/nfs/speed-scratch/qiaoyu/speed-hpc/project/TRELLIS/datasets/Toys4k_small/gaussians_decoded_vis_weighted_abo_ft_step170k" \
  --resolution 512 \
  --bg 0,0,0
